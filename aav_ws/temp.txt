cat << 'EOF' > install_jetson_torch.sh
#!/usr/bin/env bash
set -e

echo "===== Jetson Auto PyTorch Installer ====="

# --- Detect Python ---
PY=$(python3 -c "import sys; print(f'{sys.version_info.major}.{sys.version_info.minor}')")
echo "Python version: $PY"

if [ "$PY" != "3.10" ]; then
  echo "❌ ERROR: JetPack 6 requires Python 3.10"
  exit 1
fi

# --- Detect JetPack ---
JP=$(dpkg -l | grep -i jetpack | awk '{print $3}' | head -n1 || true)
echo "JetPack version: ${JP:-unknown}"

# --- Clean installs ---
echo "Cleaning old installs..."
pip uninstall -y torch torchvision ultralytics || true
rm -rf ~/.local/lib/python3.10/site-packages/torch*
rm -rf ~/.local/lib/python3.10/site-packages/torchvision*
pip cache purge || true

# --- Install deps ---
echo "Installing dependencies..."
pip install --user "numpy<2"

# --- Install correct wheels (JP6 CUDA 12.6 known working combo) ---
echo "Downloading PyTorch + TorchVision..."
wget -O torch.whl https://pypi.jetson-ai-lab.dev/jp6/cu126/+f/6cc/6ecfe8a5994fd/torch-2.6.0-cp310-cp310-linux_aarch64.whl
wget -O torchvision.whl https://pypi.jetson-ai-lab.dev/jp6/cu126/+f/aa2/2da8dcf4c4c8d/torchvision-0.21.0-cp310-cp310-linux_aarch64.whl

echo "Installing PyTorch..."
pip install --user --no-cache-dir torch.whl

echo "Installing TorchVision..."
pip install --user --no-cache-dir torchvision.whl

# --- Install YOLO ---
echo "Installing Ultralytics..."
pip install --user ultralytics

# --- Verify ---
echo "Verifying install..."
python3 - <<PYTHON
import torch
print("Torch version:", torch.__version__)
print("CUDA available:", torch.cuda.is_available())
PYTHON

echo "===== INSTALL COMPLETE ====="
EOF

chmod +x install_jetson_torch.sh
./install_jetson_torch.sh
