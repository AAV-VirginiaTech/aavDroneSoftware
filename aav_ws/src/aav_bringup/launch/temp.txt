#!/usr/bin/env bash
set -eo pipefail

# =========================
# User-editable settings
# =========================
ROS_DISTRO="humble"
NET_IFACE="eth0"
JETSON_IP="192.168.144.30/24"
CAMERA_IP="192.168.144.25"
RTSP_URL="rtsp://${CAMERA_IP}:8554/main.264"
CONFIG_DIR="${HOME}/siyi_a8/config"
CONFIG_FILE="${CONFIG_DIR}/a8_mini_gscam.yaml"
IMAGE_TOPIC="/siyi_a8/image_raw"
CAMINFO_TOPIC="/siyi_a8/camera_info"

echo "=== SIYI A8 Mini ROS 2 startup ==="

# =========================
# Basic checks
# =========================
if [ ! -f "/opt/ros/${ROS_DISTRO}/setup.bash" ]; then
  echo "ROS 2 setup file not found at /opt/ros/${ROS_DISTRO}/setup.bash"
  exit 1
fi

if ! command -v gst-inspect-1.0 >/dev/null 2>&1; then
  echo "gst-inspect-1.0 not found. Install GStreamer first."
  exit 1
fi

# =========================
# Check required GStreamer plugins
# =========================
for plugin in rtspsrc rtph265depay h265parse nvv4l2decoder nvvidconv videoconvert; do
  if ! gst-inspect-1.0 "$plugin" >/dev/null 2>&1; then
    echo "Missing GStreamer plugin: $plugin"
    exit 1
  fi
done

# =========================
# Configure network interface
# =========================
echo "Configuring ${NET_IFACE} as ${JETSON_IP} ..."
sudo ip addr flush dev "${NET_IFACE}" || true
sudo ip addr add "${JETSON_IP}" dev "${NET_IFACE}"
sudo ip link set "${NET_IFACE}" up

echo "Pinging camera at ${CAMERA_IP} ..."
if ! ping -c 3 "${CAMERA_IP}"; then
  echo "Camera did not respond to ping."
  exit 1
fi

# =========================
# Write gscam config
# =========================
mkdir -p "${CONFIG_DIR}"

cat > "${CONFIG_FILE}" <<EOF
/**:
  ros__parameters:
    gscam_config: "rtspsrc location=${RTSP_URL} protocols=tcp latency=100 ! rtph265depay ! h265parse ! nvv4l2decoder ! nvvidconv ! videoconvert ! video/x-raw,format=RGB"
    camera_name: "a8_mini"
    frame_id: "a8_mini_optical_frame"
    image_encoding: "rgb8"
    use_gst_timestamps: true
    sync_sink: false
    use_sensor_data_qos: true
EOF

echo "Wrote config to ${CONFIG_FILE}"

# =========================
# Source ROS 2 safely
# =========================
set +u
source "/opt/ros/${ROS_DISTRO}/setup.bash"
set -u

# If you later build your own workspace, uncomment and adjust:
# set +u
# source "${HOME}/ros2_ws/install/setup.bash"
# set -u

# =========================
# Launch gscam
# =========================
echo "Starting gscam..."
exec ros2 run gscam gscam_node \
  --ros-args \
  --params-file "${CONFIG_FILE}" \
  -r /camera/image_raw:="${IMAGE_TOPIC}" \
  -r /camera/camera_info:="${CAMINFO_TOPIC}"
